<section class="landing-section">
  <div class="section-header">
    <h2 class="section-title">A Discord bot in twenty lines</h2>
    <p class="section-desc">Initialize DisTube with your discord.js client and handle message commands.</p>
  </div>

```js
import { DisTube, Events } from "distube";
import { Client, GatewayIntentBits } from "discord.js";

const client = new Client({
  intents: [
    GatewayIntentBits.Guilds,
    GatewayIntentBits.GuildVoiceStates,
    GatewayIntentBits.GuildMessages,
    GatewayIntentBits.MessageContent,
  ],
});

const distube = new DisTube(client);

distube.on(Events.PLAY_SONG, (queue, song) => {
  queue.textChannel.send(`Playing: ${song.name} (${song.formattedDuration})`);
});

client.on("messageCreate", async (message) => {
  if (!message.content.startsWith("!play")) return;
  const query = message.content.slice(6).trim();
  if (!query) return;

  await distube.play(message.member.voice.channel, query, {
    message,
    member: message.member,
    textChannel: message.channel,
  });
});

client.login(process.env.DISCORD_TOKEN);
```

    ```js

    import { DisTube } from "distube";

    const distube = new DisTube(client, {
      plugins: [new SpotifyPlugin()],
    });
    ```
  </TabItem>
  <TabItem label="Custom extractor">
    ```js
    import { ExtractorPlugin } from "distube";

    class CustomExtractor extends ExtractorPlugin {
      async validate(url) {
        return url.startsWith("https://audio.example.com/");
      }
      async resolve(url, options) {
        // Return a Song or Playlist
      }
      async getStreamURL(song) {
        return song.url;
      }
    }
    ```
